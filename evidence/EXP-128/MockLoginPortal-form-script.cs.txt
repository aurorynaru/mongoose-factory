using System;
using System.Collections.Generic;
using Mongoose.Scripting;
using Newtonsoft.Json;

//<Ref>Newtonsoft.Json.dll</Ref>

namespace Mongoose.FormScripts
{
    public class MockLoginPortal : FormScript
    {
        // -----------------------------------------------------------------
        // JS passes the credentials here; C# decides and alerts.
        // JS: WSForm.invoke("ProcessLogin", username, password, callback)
        // -----------------------------------------------------------------
        public string ProcessLogin(string username, string password)
        {
            try
            {
                if (string.IsNullOrEmpty(username) || string.IsNullOrEmpty(password))
                {
                    ShowMongooseAlert("Username and password are required.");
                    return "missing";
                }

                string properties = "Username, Password, Store, Role";
                string filter = "Username = '" + username.Replace("'", "''") + "'";

                var response = IDOClient.LoadCollection(
                    "MockLogin",
                    properties,
                    filter,
                    "",
                    1
                );

                // ---------- nothing came back ----------
                if (response.Items.Count == 0)
                {
                    ShowMongooseAlert(
                        "No data found. No user matches '" + username + "'."
                    );

                    return "empty";
                }

                // ---------- data was gotten ----------
                var item = response.Items[0];

                string storedPassword = Convert.ToString(item.PropertyValues[1].Value);
                string store          = Convert.ToString(item.PropertyValues[2].Value);
                string role           = Convert.ToString(item.PropertyValues[3].Value);

                var record = new Dictionary<string, object>();
                record["Username"] = Convert.ToString(item.PropertyValues[0].Value);
                record["Password"] = storedPassword;
                record["Store"]    = store;
                record["Role"]     = role;

                string json = JsonConvert.SerializeObject(record, Formatting.Indented);

                ThisForm.Variables("vMockLoginJson").Value = json;

                if (storedPassword != password)
                {
                    ShowMongooseAlert("Data received, but the password is incorrect.");
                    return "invalid";
                }

                ShowMongooseAlert("Data received:\n" + json);
                return "ok";
            }
            catch (Exception ex)
            {
                // ---------- thrown ----------
                ShowMongooseAlert("Login error: " + ex.Message);
                return "error";
            }
        }

        // -----------------------------------------------------------------
        // Single alert helper — if this one line doesn't compile in your
        // version, swap it for the alternative shown in the comment.
        // -----------------------------------------------------------------
        private void ShowMongooseAlert(string message)
        {
            // Swap this ONE line:
            // ThisForm.ShowMessage(message);
            Application.ShowMessage(message);
            // ThisForm.ShowMessage("Mongoose Login", message);
        }

        // -----------------------------------------------------------------
        // Your original loader — kept, unchanged.
        // -----------------------------------------------------------------
        public string LoadMockLoginJson()
        {
            string idoName = "MockLogin";
            string properties = "Username, Password, Store, Role";

            try
            {
                var response = IDOClient.LoadCollection(
                    idoName,
                    properties,
                    "",
                    "",
                    100
                );

                var users = new List<Dictionary<string, object>>();

                for (int i = 0; i < response.Items.Count; i++)
                {
                    var item = response.Items[i];
                    var record = new Dictionary<string, object>();

                    record["Username"] = item.PropertyValues[0].Value;
                    record["Password"] = item.PropertyValues[1].Value;
                    record["Store"] = item.PropertyValues[2].Value;
                    record["Role"] = item.PropertyValues[3].Value;

                    users.Add(record);
                }

                string json = JsonConvert.SerializeObject(users, Formatting.Indented);

                ThisForm.Variables("vMockLoginJson").Value = json;

                return json;
            }
            catch (Exception ex)
            {
                var error = new Dictionary<string, object>();
                error["error"] = ex.ToString();
                return JsonConvert.SerializeObject(error, Formatting.Indented);
            }
        }
    }
}
