Public Sub ParseResponseHeaders(ByVal headerString as String, ByVal headerCompName as String)
   If Not String.IsNullOrEmpty(headerCompName) And Not ThisForm.Components(headerCompName) is Nothing
      dim comp as IWSFormComponent = ThisForm.Components(headerCompName)
      comp.Value = String.Empty
      dim jsonReader as JsonTextReader = New JsonTextReader( new StringReader( headerstring ) )
      while jsonReader.Read()
         select case jsonReader.TokenType
            case JsonToken.StartArray, JsonToken.StartObject, JsonToken.EndArray, JsonToken.EndObject
               continue while
            case JsonToken.PropertyName
               comp.Value += String.Format("{0}:", jsonReader.Value)
            case Else
               comp.Value += String.Format(" {0}{1}", jsonReader.Value, Environment.NewLine)
         end select
      end while
      jsonReader.Close()
      'This call requires reflection, which doesn't work in partial trust. Above code using JsonTextReader should work
      'dim parsedHeaders() as Tuple(of string, string) = JsonConvert.DeserializeObject(Of Tuple(Of string, string)())(headerString)
      'for each header as Tuple(of string, string) in parsedHeaders
      '   Application.ShowMessage(header.ToString())
      '   comp.Value += String.Format("{0}: {1}{2}", header.Item1, header.Item2, Environment.NewLine)
      'next
   Else
      Application.ShowMessage(string.Format("Invalid component name: {0}", headerCompName))
   End If
End Sub
